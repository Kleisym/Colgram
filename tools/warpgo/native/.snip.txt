// PROTOCOL_ERROR for :protocol and wants cf-connect-proto instead, which is what its own client sends.
type h2CapsuleStream struct {
	h2 *h2raw
}

func (cs *h2CapsuleStream) SendDatagram(data []byte) error {
	// One zero byte in front of the packet and nothing else. The generalised capsule with a type, a
	// length and a context id was measured on this edge and answered with silence - the stream stayed
	// open and not one frame came back - while this short form is what the working client writes.
	return cs.h2.writeData(append([]byte{0x00}, data...))
}
